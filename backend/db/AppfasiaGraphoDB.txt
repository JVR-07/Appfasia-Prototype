/* 1. Crear tipos de Vértices (Vertex Types) */
CREATE VERTEX TYPE Nivel;
CREATE VERTEX TYPE Hito;

/* 2. Crear tipos de Aristas (Edge Types) */
-- Define que un hito pertenece a un nivel específico
CREATE EDGE TYPE PERTENECE_A;
-- Define que un hito debe completarse antes que otro (Prerrequisito)
CREATE EDGE TYPE PRERREQUISITO_DE;
-- Relaciona un hito con un recurso atómico de la BD Relacional (ID externo)
CREATE EDGE TYPE USA_RECURSO;

/* 3. Definir Propiedades e Índices */
CREATE PROPERTY Hito.id_hito STRING; -- Coincide con id_hito_grafo en tu tabla Estado_hito
CREATE PROPERTY Hito.descripcion STRING;
CREATE PROPERTY Hito.edad_minima INTEGER;
CREATE INDEX ON Hito (id_hito) UNIQUE;

CREATE PROPERTY Nivel.numero INTEGER;
CREATE PROPERTY Nivel.nombre STRING;
CREATE INDEX ON Nivel (numero) UNIQUE;

/* Insertar Niveles de Dificultad [cite: 852-857] */
INSERT INTO Nivel SET numero = 1, nombre = 'Descubrimiento Lingüístico', edad_rango = '0-2';
INSERT INTO Nivel SET numero = 2, nombre = 'Estructuración Telegráfica', edad_rango = '2-4';

/* Insertar Hitos Ejemplares (Nivel 1: 0-1 año) [cite: 1145-1155] */
INSERT INTO Hito SET id_hito = 'H01_REP_SON', descripcion = 'Repite sonidos que hacen otros', edad_minima = 0;
INSERT INTO Hito SET id_hito = 'H01_REP_SIL', descripcion = 'Repite la misma sílaba 2 o 3 veces', edad_minima = 0;
INSERT INTO Hito SET id_hito = 'H01_COMB_SIL', descripcion = 'Combina dos sílabas distintas', edad_minima = 1;

/* Crear Relaciones de Pertenencia al Nivel 1 */
CREATE EDGE PERTENECE_A FROM (SELECT FROM Hito WHERE id_hito = 'H01_REP_SON') TO (SELECT FROM Nivel WHERE numero = 1);
CREATE EDGE PERTENECE_A FROM (SELECT FROM Hito WHERE id_hito = 'H01_REP_SIL') TO (SELECT FROM Nivel WHERE numero = 1);
CREATE EDGE PERTENECE_A FROM (SELECT FROM Hito WHERE id_hito = 'H01_COMB_SIL') TO (SELECT FROM Nivel WHERE numero = 1);

/* Crear Prerrequisitos (Lógica de Navegación ) */
-- Para combinar sílabas, primero debe poder repetir una misma sílaba
CREATE EDGE PRERREQUISITO_DE 
FROM (SELECT FROM Hito WHERE id_hito = 'H01_REP_SIL') 
TO (SELECT FROM Hito WHERE id_hito = 'H01_COMB_SIL');

/* Relacionar un Hito con un Recurso Atómico (id_recurso de tu tabla SQL) */
CREATE EDGE USA_RECURSO FROM (SELECT FROM Hito WHERE id_hito = 'H01_REP_SIL') 
TO (SELECT FROM Hito WHERE id_hito = 'H01_REP_SIL') -- En un grafo puro podrías crear nodos recurso, 
-- pero aquí guardamos el ID relacional como propiedad en el hito:
UPDATE Hito SET sql_recursos_ids = [1, 5, 12] WHERE id_hito = 'H01_REP_SIL';

/* Buscar hitos usando el operador de instancia */
SELECT descripcion FROM (
  TRAVERSE out("PRERREQUISITO_DE") FROM (SELECT FROM Hito WHERE id_hito = 'H01_REP_SIL')
) WHERE @this INSTANCEOF 'Hito';
