-- Entidad usuario 
CREATE TABLE Usuario
(
   id_usuario SERIAL,
   email VARCHAR(255) UNIQUE NOT NULL,
   password_hash VARCHAR(255) NOT NULL,
   rol VARCHAR(20) NOT NULL,
   fecha_registro TIMESTAMP DEFAULT NOW(),
   ultimo_acceso TIMESTAMP,
   activo BOOLEAN DEFAULT TRUE,
   CONSTRAINT pk_id_usuario PRIMARY KEY(id_usuario)
);

-- Entidad perfil del niño
CREATE TABLE Perfil_nino
(
   id_perfil_nino SERIAL,
   id_usuario INT NOT NULL,
   nombre VARCHAR(100) NOT NULL,
   fecha_nacimiento DATE NOT NULL,
   edad_cronologica_meses INT,
   nivel_actual VARCHAR(20) NOT NULL,
   ultimo_ime DECIMAL(4,2),
   ultimo_ipf DECIMAL(5,2),
   ultimo_tra DECIMAL(5,2),
   avatar_seleccionado VARCHAR(50),
   fecha_ultima_sesion TIMESTAMP,
   suspension_temporal_vm BOOLEAN DEFAULT FALSE,
   CONSTRAINT pk_id_perfil_nino PRIMARY KEY(id_perfil_nino),
   CONSTRAINT fk_id_usuario_perfil_nino FOREIGN KEY(id_usuario) REFERENCES Usuario(id_usuario)
);

-- Entidad perfil del padre
CREATE TABLE Perfil_padre
(
   id_perfil_padre SERIAL,
   id_usuario INT NOT NULL,
   telefono VARCHAR(20) NOT NULL,
   alertas_activadas BOOLEAN DEFAULT TRUE,
   idioma_preferido VARCHAR(10) DEFAULT 'es',
   CONSTRAINT pk_id_padre PRIMARY KEY(id_perfil_padre),
   CONSTRAINT fk_id_usuario_id_padre FOREIGN KEY(id_usuario) REFERENCES Usuario(id_usuario) 
);

-- Entidad de supervision
CREATE TABLE Supervision
(
   id_supervision SERIAL,
   id_perfil_padre INT NOT NULL,
   id_perfil_nino INT NOT NULL,
   es_tutor_principal BOOLEAN DEFAULT TRUE,
   fecha_asignacion TIMESTAMP DEFAULT NOW(),
   CONSTRAINT pk_id_supervision PRIMARY KEY(id_supervision),
   CONSTRAINT fk_id_supervision_padre FOREIGN KEY(id_perfil_padre) REFERENCES Perfil_padre (id_perfil_padre),
   CONSTRAINT fk_id_supervision_nino FOREIGN KEY(id_perfil_nino) REFERENCES Perfil_nino(id_perfil_nino)
);

-- Entidad de categorias semanticas
CREATE TABLE Categoria_semantica
(
   id_categoria SERIAL,
   nombre VARCHAR(100) UNIQUE NOT NULL,
   nivel_minimo INT NOT NULL DEFAULT 1,
   descripcion TEXT,
   CONSTRAINT pk_id_categoria PRIMARY KEY(id_categoria)
);

-- Entidad de los fonemas
CREATE TABLE Fonema
(
   id_fonema SERIAL,
   representacion VARCHAR(10) UNIQUE NOT NULL,
   dificultad VARCHAR(20) NOT NULL DEFAULT 'baja',
   nivel_aparicion INT NOT NULL DEFAULT 1,
   CONSTRAINT pk_id_fonema PRIMARY KEY(id_fonema)
);

-- Entidad de los recursos atomicos
CREATE TABLE Recurso_atomico
(
   id_recurso SERIAL,
   tipo VARCHAR(20) NOT NULL DEFAULT 'palabra',
   texto TEXT,
   nivel_segurido INT NOT NULL DEFAULT 1,
   fonema_objetivo_id INT NULL,
   dificultad_articulacion VARCHAR(20) NOT NULL DEFAULT 'baja',
   url_media TEXT,
   metadata_json JSONB, -- Corregido JOSNB a JSONB
   CONSTRAINT pk_id_recurso PRIMARY KEY(id_recurso),
   CONSTRAINT fK_id_fonema_recurso FOREIGN KEY(fonema_objetivo_id) REFERENCES Fonema(id_fonema)
);

-- Entidad de recurso categoria
CREATE TABLE Recurso_categoria
(
   id_recurso_categoria SERIAL,
   id_recurso INT NOT NULL,
   id_categoria INT NOT NULL,
   peso_relevancia DECIMAL(3,2) DEFAULT 1.0,
   CONSTRAINT pk_id_recurso_categoria PRIMARY KEY(id_recurso_categoria),
   CONSTRAINT fk_id_recurso_atomico_categoria FOREIGN KEY(id_recurso) REFERENCES Recurso_atomico (id_recurso),
   CONSTRAINT fk_id_categoria_semantica_recurso FOREIGN KEY(id_categoria) REFERENCES Categoria_semantica(id_categoria)
);

-- Entidad de plantillas para los ejercicios
CREATE TABLE Plantilla_ejercicios
(
   id_plantilla SERIAL,
   nombre VARCHAR(100) NOT NULL,
   hardware_req VARCHAR(10) NOT NULL DEFAULT 'V-M',
   objetivo_clinico VARCHAR(20) NOT NULL DEFAULT 'IPF',
   configuracion_base JSONB NOT NULL,
   feedback_exito TEXT,
   feedback_reintento TEXT,
   feedback_timeout  TEXT,
   requiere_calibracion_microfono BOOLEAN DEFAULT FALSE,
   CONSTRAINT pk_id_plantilla PRIMARY KEY(id_plantilla) 
);

-- Entidad para los minijuegos
CREATE TABLE Minijuego
(
   id_minijuego SERIAL,
   nombre VARCHAR(100) NOT NULL DEFAULT 'juego aprendizaje',
   hardware_req VARCHAR(10) NOT NULL DEFAULT 'V-M',
   nivel_minimo INT NOT NULL DEFAULT 1,
   duracion_segundos INT NOT NULL DEFAULT 1,
   configuracion JSONB,
   url_asset TEXT,
   CONSTRAINT pk_id_minijuego PRIMARY KEY(id_minijuego)
);

-- Entidad de session
CREATE TABLE Sesion
(
   id_sesion SERIAL,
   id_perfil_nino INT NOT NULL,
   titulo VARCHAR(200) NOT NULL,
   narrativa TEXT,
   fecha_programacion TIMESTAMP,
   fecha_inicio TIMESTAMP NULL,
   fecha_fin TIMESTAMP NULL,
   duracion_segundos INT, -- Corregido nombre y tipo
   estado VARCHAR(20) NOT NULL DEFAULT 'programada',
   nivel_dificultad INT NOT NULL DEFAULT 1,
   es_recuperacion  BOOLEAN DEFAULT FALSE,
   CONSTRAINT pk_id_sesion PRIMARY KEY(id_sesion),
   CONSTRAINT fk_peril_nino_sesion FOREIGN KEY(id_perfil_nino) REFERENCES Perfil_nino(id_perfil_nino)
);

-- Entidad de instancia de la actividad
CREATE TABLE Instancia_actividad
(
   id_instancia SERIAL,
   id_sesion INT NOT NULL,
   id_plantilla INT NOT NULL, -- Corregido: faltaba tipo INT
   orden INT NOT NULL,
   es_minijuego BOOLEAN DEFAULT FALSE,
   id_minijuego INT NULL,
   recursos_json JSONB,
   intentos_permitidos INT DEFAULT 3,
   configuracion_local JSONB,
   estado VARCHAR(20) DEFAULT 'pendiente',
   umbral_exito_ipf INT DEFAULT 80,
   tiempo_max_tra_segundos DECIMAL(5,2) DEFAULT 5.0,
   CONSTRAINT pk_id_instancia PRIMARY KEY(id_instancia),
   CONSTRAINT fk_id_sesion_instancia FOREIGN KEY(id_sesion) REFERENCES Sesion(id_sesion),
   CONSTRAINT fk_id_plantilla_instancia FOREIGN KEY(id_plantilla) REFERENCES Plantilla_ejercicios(id_plantilla),
   CONSTRAINT fk_id_minijuego_instancia FOREIGN KEY(id_minijuego) REFERENCES Minijuego(id_minijuego) -- Corregido FOREGIN
);

-- Entidad de respuesta de los ejercicios
CREATE TABLE Respuesta_ejercicio
(
   id_respuesta SERIAL,
   id_instancia INT NOT NULL,
   ipf_obtenido DECIMAL(5,2),
   tra_segundos DECIMAL(6,3),
   intento_numero INT NOT NULL DEFAULT 1,
   transcripcion_voz TEXT,
   respuesta_usuario TEXT,
   fue_correcto BOOLEAN,
   aplico_ayuda_asistente BOOLEAN DEFAULT FALSE,
   fue_timeout BOOLEAN DEFAULT FALSE,
   timestamp_respuesta TIMESTAMP DEFAULT NOW(),
   CONSTRAINT pk_id_respuesta PRIMARY KEY(id_respuesta),
   CONSTRAINT fk_id_instancia_respuesta FOREIGN KEY(id_instancia) REFERENCES Instancia_actividad(id_instancia)
);

-- Entidad de la metrica de la session
CREATE TABLE Metrica_sesion
(
    id_metrica SERIAL,
    id_sesion INT NOT NULL UNIQUE,
    ime_calculado DECIMAL(4,2),
    ipf_promedio DECIMAL(5,2),
    tra_promedio_segundos DECIMAL(6,3),
    total_ejercicios INT,
    total_aciertos INT,
    total_timeouts INT,
    total_ayudas INT,
    porcentaje_maestria DECIMAL(5,2),
    desglose_hitos_json JSONB,
    recomendacion_siguiente TEXT,
    CONSTRAINT pk_id_metrica PRIMARY KEY(id_metrica),
    CONSTRAINT fk_id_metrica_sesion FOREIGN KEY(id_sesion) REFERENCES Sesion(id_sesion)
);

-- Entidad de los estados de los hitos
CREATE TABLE Estado_hito
(
    id_estado_hito SERIAL,
    id_perfil_nino INT NOT NULL,
    id_hito_grafo VARCHAR(100) NOT NULL,
    probabilidad_maestria DECIMAL(5,2),
    ultima_evaluacion TIMESTAMP,
    nivel_repeticion_espaciada INT,
    fecha_proxima_review DATE,
    CONSTRAINT pk_id_estado_hito PRIMARY KEY(id_estado_hito),
    CONSTRAINT fk_id_perfil_nino_hito FOREIGN KEY(id_perfil_nino) REFERENCES Perfil_nino(id_perfil_nino)
);

-- Entidad de tickets rezagados
CREATE TABLE Ticket_rezago
(
    id_ticket SERIAL,
    id_perfil_nino INT NOT NULL,
    id_hito_grafo VARCHAR(100) NOT NULL,
    tipo VARCHAR(30) NOT NULL DEFAULT 'laguna_aislada',
    nivel_detectado INT DEFAULT 1,
    fecha_deteccion TIMESTAMP DEFAULT NOW(),
    estado VARCHAR(20) DEFAULT 'pendiente',
    fecha_resolucion TIMESTAMP NULL,
    ruta_recuperacion_json  JSONB,
    CONSTRAINT pk_id_ticket PRIMARY KEY(id_ticket),
    CONSTRAINT fk_id_perfil_nino_ticket FOREIGN KEY(id_perfil_nino) REFERENCES Perfil_nino(id_perfil_nino)
);

-- Entidad de intervencion de tutor
CREATE TABLE Intervencion_tutor
(
    id_intervencion SERIAL,
    id_perfil_nino INT NOT NULL,
    id_perfil_padre INT NOT NULL,
    etiqueta VARCHAR(30) NOT NULL DEFAULT 'Enfermedad',
    comentario TEXT,
    ponderacion_applied DECIMAL(3,2),
    afecta_ponderacion_RO BOOLEAN DEFAULT TRUE,
    fecha_intervencion TIMESTAMP DEFAULT NOW(),
    CONSTRAINT pk_id_intervencion PRIMARY KEY(id_intervencion),
    CONSTRAINT fk_id_intervencion_nino FOREIGN KEY(id_perfil_nino) REFERENCES Perfil_nino(id_perfil_nino),
    CONSTRAINT fk_id_intervencion_padre FOREIGN KEY(id_perfil_padre) REFERENCES Perfil_padre(id_perfil_padre)
);

-- Entidad para documentacion para padres
CREATE TABLE Articulo_padres
(
    id_articulo SERIAL,
    titulo VARCHAR(200) NOT NULL,
    clasificacion VARCHAR(50) NOT NULL DEFAULT 'estrategias',
    contenido_texto TEXT NOT NULL,
    url_infografia TEXT,
    url_video TEXT,
    fuente_bibliografica VARCHAR(500),
    nivel_minimo_recomendado INT DEFAULT 4,
    fecha_publicacion DATE,
    activo BOOLEAN DEFAULT TRUE,
    CONSTRAINT pk_id_articulo PRIMARY KEY(id_articulo)
);

-- Entidad de visualizacion del articulo 
CREATE TABLE Visualizacion_articulo
(
    id_visualizacion SERIAL,
    id_articulo INT NOT NULL,
    id_perfil_padre INT NOT NULL,
    fecha_visto TIMESTAMP DEFAULT NOW(),
    recibio_alerta_asociada BOOLEAN DEFAULT FALSE,
    tiempo_lectura_segundos INT,
    CONSTRAINT pk_id_visualizacion PRIMARY KEY(id_visualizacion),
    CONSTRAINT fk_id_visualizacion_articulo FOREIGN KEY(id_articulo) REFERENCES Articulo_padres(id_articulo),
    CONSTRAINT fk_id_visualizacion_padre FOREIGN KEY(id_perfil_padre) REFERENCES Perfil_padre(id_perfil_padre)   
);

-- Entidad de alertas tempranas
CREATE TABLE Alerta_temprana
(
    id_alerta SERIAL,
    id_perfil_nino INT NOT NULL,
    tipo_alerta VARCHAR(50) NOT NULL DEFAULT 'ausencia_balbuceo',
    severidad VARCHAR(20) DEFAULT 'informativa',
    mensaje TEXT,
    fecha_generacion TIMESTAMP DEFAULT NOW(),
    fecha_envio TIMESTAMP NULL,
    atendida BOOLEAN DEFAULT FALSE,
    id_articulo_asociado INT NULL,
    CONSTRAINT pk_id_alerta PRIMARY KEY(id_alerta),
    CONSTRAINT fk_id_perfil_nino FOREIGN KEY(id_perfil_nino) REFERENCES Perfil_nino(id_perfil_nino),
    CONSTRAINT fk_articulo_alerta_temprana FOREIGN KEY(id_articulo_asociado) REFERENCES Articulo_padres(id_articulo)
);

-- Entidad para los logs y auditoria
CREATE TABLE Log_sistema
(
    id_log BIGSERIAL,
    nivel VARCHAR(20) NOT NULL DEFAULT 'INFO',
    origen VARCHAR(100),
    mensaje TEXT,
    metadata_json JSONB,
    timestamp TIMESTAMP DEFAULT NOW(),
    CONSTRAINT pk_id_log PRIMARY KEY(id_log)
);

-- Busca recursos por nivel y tipo (ej. Palabras de nivel 2)
CREATE INDEX idx_recurso_nivel_tipo ON Recurso_atomico (nivel_segurido, tipo);

-- Relaciona rápidamente los recursos con sus fonemas objetivo
CREATE INDEX idx_recurso_fonema ON Recurso_atomico (fonema_objetivo_id);

-- Acelera la búsqueda de palabras por categoría semántica (ej. "Frutas")
CREATE INDEX idx_recurso_categoria_rel ON Recurso_categoria (id_categoria, id_recurso);

-- Vital para cargar las últimas sesiones del niño y calcular el EMA (Promedio Móvil)
CREATE INDEX idx_sesion_nino_fecha ON Sesion (id_perfil_nino, fecha_inicio DESC);

-- Relaciona las respuestas con la instancia de actividad para análisis en tiempo real
CREATE INDEX idx_respuesta_instancia ON Respuesta_ejercicio (id_instancia);

-- Para verificar el progreso en el Grafo de Conocimiento (Hitos)
CREATE INDEX idx_estado_hito_nino ON Estado_hito (id_perfil_nino, id_hito_grafo);

-- Para las etiquetas de "Cansancio" o "Enfermedad" que afectan la ponderación
CREATE INDEX idx_intervencion_nino_padre ON Intervencion_tutor (id_perfil_nino, id_perfil_padre);

-- Para mostrar alertas pendientes (Ej. Alerta de "ausencia de balbuceo")
CREATE INDEX idx_alerta_nino_pendiente ON Alerta_temprana (id_perfil_nino) WHERE atendida = FALSE;

-- Para el seguimiento de qué artículos han leído los padres
CREATE INDEX idx_visualizacion_padre ON Visualizacion_articulo (id_perfil_padre, id_articulo);

-- Permite buscar dentro de los campos JSON de los logs de forma eficiente
CREATE INDEX idx_log_metadata_gin ON Log_sistema USING GIN (metadata_json);
